### Step 2 — If reachable, build source client

Create `scripts/ingest/sources/theyworkforyou.ts`:

```typescript
// TheyWorkForYou API — Hansard debates
// Docs: https://www.theyworkforyou.com/api/docs/getDebates
// Rate limit: 1 req/sec for free tier, more with key
// Coverage: Commons + Lords from ~1988, Westminster Hall from 1999

const TWFY_BASE = 'https://www.theyworkforyou.com/api'
// API key from env: process.env.TWFY_API_KEY
// Register at https://www.theyworkforyou.com/api/key — free for non-commercial/civic use

export async function fetchHansardByDate(
  type: 'commons' | 'lords' | 'westminhall',
  date: string // YYYY-MM-DD
): Promise<string> {
  const url = `${TWFY_BASE}/getDebates?type=${type}&date=${date}&output=json&key=${process.env.TWFY_API_KEY}`
  const res = await fetch(url)
  if (!res.ok) throw new Error(`TWFY ${type} ${date}: HTTP ${res.status}`)
  const data = await res.json()
  // data.rows = array of debate sections
  // Each row has: body (text), speaker, subject, date
  return data.rows?.map((r: any) => r.body).filter(Boolean).join('\n\n') ?? ''
}

export async function countDebatesByType(type: 'commons' | 'lords' | 'westminhall'): Promise<number> {
  // TWFY doesn't have a count endpoint — estimate from date range
  // Commons: ~250 sitting days/year × ~35 years = ~8,750 date chunks
  // Use calendar approach: generate all sitting days and check
  return type === 'commons' ? 8_750 : type === 'lords' ? 7_000 : 3_000
}
```

Add `TWFY_API_KEY` to Railway env vars for the scheduler and workers.
Add to `source_rate_limits`: `sourceKey = 'theyworkforyou'`, `intervalMs = 1000`.
