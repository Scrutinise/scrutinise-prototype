### Step 2 — Build source client `scripts/ingest/sources/twfy-pwdata.ts`
```typescript
// TWFY pwdata — bulk Hansard XML, freely accessible
// One file per sitting day, format: debates{YYYY-MM-DD}.xml
// Coverage: Commons 1919+, Lords 2001+, WH 2001+, Written Answers 2001+

const BASE = 'https://www.theyworkforyou.com/pwdata/scrapedxml'

export type PwdataType = 'debates' | 'lords' | 'westminster' | 'wrans' | 'lordswrans'

export async function fetchPwdataFile(
  type: PwdataType,
  date: string // YYYY-MM-DD
): Promise<string | null> {
  const filename = `${type}${date}.xml`
  const url = `${BASE}/${type}/${filename}`
  const res = await fetch(url, {
    headers: { 'User-Agent': 'Scrutinise/1.0 (civic-tech; contact@scrutinise.org)' }
  })
  if (res.status === 404) return null // No sitting that day
  if (!res.ok) throw new Error(`pwdata ${type} ${date}: HTTP ${res.status}`)
  return await res.text()
}

export function parsePwdataXml(xml: string): string {
  // Extract text from <speech> elements
  // Each speech has: person_id, speakername, oral-qnum (if oral question), body
  // <speech ...><p>Text here</p></speech>
  const speeches: string[] = []
  const speechRegex = /<speech[^>]*>([\s\S]*?)<\/speech>/g
  let match
  while ((match = speechRegex.exec(xml)) !== null) {
    // Strip XML tags from speech content
    const text = match[1].replace(/<[^>]+>/g, ' ').replace(/\s+/g, ' ').trim()
    if (text.length > 20) speeches.push(text)
  }
  return speeches.join('\n\n')
}

export async function listPwdataDates(type: PwdataType): Promise<string[]> {
  // Fetch directory listing and extract dates
  const res = await fetch(`${BASE}/${type}/`)
  const html = await res.text()
  const pattern = new RegExp(`${type}(\\d{4}-\\d{2}-\\d{2})\\.xml`, 'g')
  const dates: string[] = []
  let match
  while ((match = pattern.exec(html)) !== null) {
    dates.push(match[1])
  }
  return dates.sort()
}
```
