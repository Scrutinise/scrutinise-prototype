## PART 2 — Migrate Railway corpus_sections to Neon

**Run after Part 1 is committed and deployed** — confirms counts read from Neon before
we move data there.

Build `scripts/ingest/migrate-corpus-to-neon.ts`:

```typescript
// Migrates all corpus_sections rows from Railway → Neon
// Idempotent: ON CONFLICT DO NOTHING
// Checkpoint/resume: saves progress to scripts/ingest/corpus-migration-checkpoint.json
// Batch size: 200 rows per insert (matches transfer-to-neon.ts pattern)

import { Pool } from 'pg'
import * as fs from 'fs'

const CHECKPOINT_FILE = path.join(__dirname, 'corpus-migration-checkpoint.json')
const BATCH_SIZE = 200

async function migrate() {
  const railwayPool = new Pool({ 
    connectionString: process.env.DATABASE_URL,
    ssl: { rejectUnauthorized: false }
  })
  const neonPool = new Pool({
    connectionString: process.env.NEON_DATABASE_URL,
    ssl: { rejectUnauthorized: false }
  })

  // Load checkpoint
  let lastId = 0
  if (fs.existsSync(CHECKPOINT_FILE)) {
    lastId = JSON.parse(fs.readFileSync(CHECKPOINT_FILE, 'utf8')).lastId
    console.log(`[migrate] resuming from id > ${lastId}`)
  }

  // Get total count
  const { rows: [{count}] } = await railwayPool.query(
    'SELECT COUNT(*) FROM corpus_sections WHERE id > $1', [lastId]
  )
  console.log(`[migrate] ${count} rows to migrate`)

  let migrated = 0
  while (true) {
    const { rows } = await railwayPool.query(
      `SELECT * FROM corpus_sections WHERE id > $1 ORDER BY id LIMIT $2`,
      [lastId, BATCH_SIZE]
    )
    if (rows.length === 0) break

    // Build bulk insert — column list must match Neon schema exactly
    // Get column names from first row
    const cols = Object.keys(rows[0])
    const placeholders = rows.map((_, i) => 
      `(${cols.map((_, j) => `$${i * cols.length + j + 1}`).join(',')})`
    ).join(',')
    const values = rows.flatMap(r => cols.map(c => r[c]))

    await neonPool.query(
      `INSERT INTO corpus_sections (${cols.map(c => `"${c}"`).join(',')})
       VALUES ${placeholders}
       ON CONFLICT (id) DO NOTHING`,
      values
    )

    lastId = rows[rows.length - 1].id
    migrated += rows.length
    fs.writeFileSync(CHECKPOINT_FILE, JSON.stringify({ lastId }))
    
    if (migrated % 10000 === 0) {
      console.log(`[migrate] ${migrated} migrated, last id: ${lastId}`)
    }
  }

  console.log(`[migrate] complete: ${migrated} rows migrated`)
  
  // Verify
  const { rows: [{ neon_count }] } = await neonPool.query(
    'SELECT COUNT(*) as neon_count FROM corpus_sections'
  )
  const { rows: [{ railway_count }] } = await railwayPool.query(
    'SELECT COUNT(*) as railway_count FROM corpus_sections'
  )
  console.log(`[migrate] Railway: ${railway_count}, Neon: ${neon_count}`)

  await railwayPool.end()
  await neonPool.end()
}

migrate().catch(e => { console.error(e); process.exit(1) })
```

## PART 3 — Census: query actual counts from Neon + R2

Build `scripts/ingest/census/live-census.ts` — runs every hourly scheduler cycle AND
on demand. Replaces estimated CORPUS_MANIFEST denominators with actuals.

```typescript
// Queries: Neon corpus_sections counts + R2 key counts by prefix
// Output: JSON written to R2 as ingest-csv/census-{date}.json
// Also returns data for the scheduler to use in email

export async function runCensus(): Promise<CensusResult> {
  // 1. Neon counts by corpus
  const neonCounts = await prismaSearch.$queryRaw`
    SELECT corpus, COUNT(*) as sections, MAX("createdAt") as latest
    FROM corpus_sections GROUP BY corpus ORDER BY sections DESC
  `
  
  // 2. Neon legacy (LegislationSection)
  const neonLegacy = await prismaSearch.$queryRaw`
    SELECT li.type, COUNT(ls.id) as sections
    FROM "LegislationItem" li
    LEFT JOIN "LegislationSection" ls ON ls."itemId" = li.id
    GROUP BY li.type ORDER BY sections DESC
  `
  
  // 3. R2 key counts by prefix (use cached counts — full R2 list is slow)
  // Cache in ingest-csv/r2-counts.json, refresh daily
  
  // 4. Queue state
  const queueState = await getPool().query(`
    SELECT corpus, status, COUNT(*) as rows,
           MAX("updatedAt") as last_activity
    FROM ingest_queue
    GROUP BY corpus, status ORDER BY corpus, status
  `)

  return { neonCounts, neonLegacy, queueState, timestamp: new Date() }
}
```

Wire `runCensus()` into the scheduler's `run()` function — run it every hour.
Use the census output to populate the email instead of hardcoded estimates.
